% Render the CSV registers in docs/registers/.
% The CSV files are the single source of truth. Edit them, then run make.
% Separator: semicolon. All TeX special characters in the CSV are printed as text.
\csvset{
  register/.style={separator=semicolon, respect all},
}

\newcommand{\entryhead}[3]{%
  \par\medskip\noindent
  \colorbox{projblue!8}{\parbox{\dimexpr\linewidth-2\fboxsep}{%
    \sffamily\bfseries #1\quad #2\hfill\status{#3}}}\par\nopagebreak}
\newcommand{\field}[2]{\par\noindent\hangindent=1.5em\hangafter=0\textsf{\textbf{#1.}} #2}
\newcommand{\csvurl}[1]{\expandafter\url\expandafter{#1}}

% ---------- Summary tables ----------
\newcommand{\gapsummary}{%
  \csvreader[register, longtable={@{}l l P{0.58\linewidth} l@{}},
    table head={\toprule \textsf{\textbf{ID}} & \textsf{\textbf{Topic}} & \textsf{\textbf{Question}} & \textsf{\textbf{Status}}\\\midrule\endhead},
    table foot={\bottomrule}]{\registerdir/gaps.csv}%
    {1=\cid,2=\ctopic,3=\cq,7=\cstatus}{\cid & \ctopic & \cq & \cstatus}}

% ---------- Full entries ----------
\newcommand{\gapentries}{%
  \csvreader[register]{\registerdir/gaps.csv}%
    {1=\cid,2=\ctopic,3=\cq,4=\cwhy,5=\ccheck,6=\csrc,7=\cstatus,8=\cres,9=\cupd}%
    {\entryhead{\cid}{\ctopic}{\cstatus}
     \field{Question}{\cq}
     \field{Why it matters}{\cwhy}
     \field{How to check}{\ccheck}
     \field{Sources}{\csrc}
     \field{Result}{\cres}
     \field{Updated}{\cupd}}}
